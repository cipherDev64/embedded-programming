printf("Thread A: Step 1 completed\n");
￼
// Update step so next thread (B) can run
step = 2;
// Wake up all waiting threads
// (Thread B will proceed, Thread C will still wait)
pthread_cond_broadcast(&cond);
// Unlock mutex after work is done
pthread_mutex_unlock(&mutex);
return NULL;
}

// This thread performs Step 2
void* threadB(void* arg)
{

pthread_mutex_lock(&mutex);
// Wait until step becomes 2
// Use while loop to recheck condition after waking up
while(step != 2)
{
printf("Thread B waiting...\n");
// This function:
// 1. Releases mutex
// 2. Puts thread to sleep
// 3. Wakes up when signaled
// 4. Re-acquires mutex automatically

printf("Thread B: Step 2 started\n");

printf("Thread B: Step 2 completed\n");

pthread_mutex_unlock(&mutex);

pthread_mutex_lock(&mutex);

{
printf("Thread C waiting...\n");
// Wait for signal from other threads
pthread_cond_wait(&cond, &mutex);
}

￼

#include <unistd.h> // fork, sleep
#include <sys/mman.h> // shm_open, mmap
#include <fcntl.h> // O_CREAT, O_RDWR
#include <sys/stat.h> // mode constants
#include <string.h> // strcpy
#include <pthread.h> // mutex
#include <sys/wait.h> // wait
#define SHM_NAME "/my_shm"
// Shared structure
typedef struct {
pthread_mutex_t mutex; // for synchronization
int data;
} shared_data_t;
int main() {
int fd;
shared_data_t *shm;
pid_t pid;
// ==============================
// STEP 1: Create Shared Memory
// ==============================
fd = shm_open(SHM_NAME, O_CREAT | O_RDWR, 0666);
if (fd == -1) {
perror("shm_open failed");
return 1;
}
// Set size of shared memory
ftruncate(fd, sizeof(shared_data_t));
// Map shared memory
shm = mmap(NULL, sizeof(shared_data_t),
PROT_READ | PROT_WRITE,
MAP_SHARED, fd, 0);
if (shm == MAP_FAILED) {
perror("mmap failed");
return 1;
}
// ==============================
// STEP 2: Initialize Mutex
// ==============================
pthread_mutexattr_t attr;
pthread_mutexattr_init(&attr);
// Make mutex shared between processes
pthread_mutexattr_setpshared(&attr, PTHREAD_PROCESS_SHARED);
pthread_mutex_init(&shm->mutex, &attr);
shm->data = 0; // initialize data
pid = fork();
if (pid == 0) {
// ==================================
// 🔹 CHILD PROCESS (WRITER)
// ==================================
sleep(1);
pthread_mutex_lock(&shm->mutex);
shm->data = 100; // write data
printf("Child: Written data = %d\n", shm->data);
pthread_mutex_unlock(&shm->mutex);
exit(0);
}
else {
// ==================================
// 🔹 PARENT PROCESS (READER)
// ==================================
sleep(2);
pthread_mutex_lock(&shm->mutex);
printf("Parent: Read data = %d\n", shm->data);
pthread_mutex_unlock(&shm->mutex);
wait(NULL);
// Cleanup
munmap(shm, sizeof(shared_data_t));
shm_unlink(SHM_NAME);
}
return 0;
}
